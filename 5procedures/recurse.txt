PROCEDURE fib(n);
BEGIN
    // WRITELN("fib", n);
    IF n <= 1 THEN
        RETURN 1;
    fib := fib(n - 1) + fib(n - 2);
END;

PROCEDURE factorial(i);
BEGIN
    // WRITELN("factorial", i);
    IF i = 0 THEN
        RETURN 1;
    ELSE
        RETURN i * factorial(i - 1);
END;

BEGIN
    FOR i := 0 TO 10 DO
    BEGIN
        WRITELN(i);
        WRITELN(fib(i));
        WRITELN(factorial(i));
        WRITELN("");
    END;
END;
.
