PROCEDURE fib(n);
BEGIN
    IF n <= 1 THEN
        RETURN 1;
    fib := fib(n - 1) + fib(n - 2);
END;

PROCEDURE factorial(i);
BEGIN
    IF i = 0 THEN factorial := 1;
    ELSE factorial := i * factorial(i - 1);
END;

BEGIN
    FOR i := 0 TO 10 DO
    BEGIN
        WRITELN(i);
        WRITELN(fib(i));
        WRITELN(factorial(i));
        WRITELN("");
    END;
END;
.
