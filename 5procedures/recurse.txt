PROCEDURE fib(n) : integer;
BEGIN
    // WRITELN("fib", n);
    IF n <= 1 THEN
        RETURN 1;
    fib := fib(n - 1) + fib(n - 2);
END;

PROCEDURE factorial(i) : integer;
BEGIN
    // WRITELN("factorial", i);
    IF i = 0 THEN
        RETURN 1;
    ELSE
        RETURN i * factorial(i - 1);
END;

PROCEDURE factorialf(i: real) : real;
BEGIN
    factorialf := 7.0;
    IF i = 0.0 THEN
        RETURN 1.0;
    ELSE
        RETURN i * factorialf(i - 1);
END;

BEGIN
    FOR i := 0 TO 10 DO
    BEGIN
        WRITELN(i);
        WRITELN(fib(i));
        WRITELN(factorialf(i * 1.0));
        WRITELN("");
    END;
END;
.
